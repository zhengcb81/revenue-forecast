        """Run the complete revenue forecast and return the published result.
    
        The result is computed as a private draft and only published after passing
        the self-contained output validator; any validation failure raises before a
        caller ever sees a publication receipt.
    
        *mode* may be ``"formal"`` (default — all hard gates must pass) or
        ``"draft"`` (unresolved data gaps are recorded as structured limitations
        but the result is returned with ``formal_output_mode="draft"``).
        invest-* consumers must only accept ``"formal"`` artifacts.
    
        Formal publications carry an ``attestation_status`` (R2.1).  I-08-B: the
        label ``"host_signed"`` is produced ONLY from a verified L3 attestation
        record (``decision.md`` section 1, rule R-LAYER-2) -- a configured provider,
        an existing file, an executable file or ``sys.executable`` are never
        sufficient.  Any provider/protocol/verification failure leaves the
        publication ``"unattested"`` with a normative failure code and no signature
        record; it fails closed and never silently claims to be signed.
    
        I-09-B: *_defer_registration* when ``True`` skips the
        ``register_publication`` call for the prepare/commit split.  The caller
        must then write all output members and call ``commit_publication()``
        explicitly.  Default ``False`` preserves backward compatibility.
        """
        from attestation_protocol import (
            attest_publication,
            payload_sha256 as attestation_payload_sha256,
            verify_publication_attestation,
        )
        from revenue_publication import (
            build_publication_receipt,
            build_draft_receipt,
            validate_publication_receipt,
        )
        from revenue_report import validate_published_forecast
    
        if mode not in {"formal", "draft"}:
            raise ValueError("mode must be 'formal' or 'draft'")
        result = _build_forecast_draft(data)
        # Phase 6 A1: validate BEFORE signing. The draft has no publication receipt
        # and no result hash; the strong validator recomputes every gate (including
        # input-gated sensitivity shocks) from the embedded input document. Only a
        # strong-validation verification context may issue a formal receipt.
        context = validate_published_forecast(result, data)
        if mode == "draft":
            result.setdefault("draft_limitations", []).append(
                "formal publication gates passed; draft mode requested"
            )
            result["publication_receipt"] = build_draft_receipt(result)
        else:
            # I-08-B step 4: ask the provider for a signature BEFORE the receipt is
            # built, so the signed payload can name the receipt's own gate set and
            # input anchor.  ``result_sha256`` / ``receipt_sha256`` are whole-artifact
            # digests that only exist after the record is attached; the design forbids
            # self-reference, so they are carried as empty commitments and the real
            # binding is ``payload_sha256`` (see decision.md).
            attestation = attest_publication(
                payload_sha256=attestation_payload_sha256(result),
                input_sha256=result["input_sha256"],
                result_sha256="",  # empty commitment; see decision.md
                receipt_sha256="",  # empty commitment; see decision.md
                gate_ids=tuple(context.executed_gate_ids),
                requested_issuer=os.environ.get(
                    "REVENUE_ATTESTATION_ISSUER", "revenue-forecast/host"
                ),
            )
            if attestation.code is not None and attestation.status != "host_signed":
                # Every signing failure fails CLOSED: the publication stays
                # ``unattested`` with the normative code recorded, and no signature
                # record is attached.  A failure must never be silently upgraded to
                # host_signed, and must never be reported as a successful signature.
                result["publication_attestation_failure"] = attestation.code
            attestation_status = (
                "host_signed" if attestation.status == "host_signed" else "unattested"
            )

                    "attestation_issuer": attestation.record["issuer"],
                    "attestation_key_id": attestation.record["key_id"],
                    "attestation_signed_at": attestation.record["signed_at"],

            # ``validated_payload_sha256`` covers the result as it will be published.
            # It lives in the design's unsigned-appendix set (see

            # attempt did, which is only knowable after the attempt, and carries no
            # trust decision of its own.
            result["publication_attestation_outcome"] = {
                "status": attestation.status,
                "failure_code": attestation.code,
                "detail": attestation.detail,
                "provider_invocations": attestation.provider_invocations,
                "handshake_ok": attestation.handshake_ok,
                "trust_domain_size": attestation.trust_domain_size,
                "W_enforced": attestation.w_enforced,
                "W_parameter": "OPEN-D7 (not adjudicated; see decision.md section 8)",

            # lock and the snapshot replay contract forbid.  The measurement belongs in
            # the operator-facing trace, not in the artifact.
            result["publication_receipt"] = build_publication_receipt(

                attestation_provenance=provenance,
